## 搜索算法

### 1. 二分查找
**适用场景**：有序数组的目标值查找，时间复杂度 O(log n)
```cpp
int binary_search(vector<int>& nums, int target) {
    int left = 0;
    int right = nums.size() - 1;
    while (left <= right) {  // 修正：left <= right 覆盖所有情况
        int mid = left + (right - left) / 2;  // 避免溢出
        if (nums[mid] > target) {
            right = mid - 1;
        } else if (nums[mid] < target) {
            left = mid + 1;
        } else {
            return mid;  // 找到目标值，返回索引
        }
    }
    return -1;  // 未找到目标值
}
```

### 2. 双指针
**适用场景**：数组元素筛选、快慢指针遍历，时间复杂度 O(n)，空间复杂度 O(1)
```cpp
class Solution {
public:
    int removeElement(vector<int>& nums, int val) {
        int slowIndex = 0;
        // 快指针遍历所有元素，慢指针记录有效元素位置
        for (int fastIndex = 0; fastIndex < nums.size(); fastIndex++) {
            if (val != nums[fastIndex]) {
                nums[slowIndex++] = nums[fastIndex];
            }
        }
        return slowIndex;  // 返回有效元素个数
    }
};
```

## 排序算法

### 1. 选择排序
**特点**：不稳定排序，原地操作，时间复杂度 O(n²)
```cpp
void selectionSort(vector<int>& nums) {
    int n = nums.size();
    // 每轮选择未排序区间的最小值
    for (int i = 0; i < n - 1; i++) {
        int minIdx = i;  // 记录最小值索引
        for (int j = i + 1; j < n; j++) {
            if (nums[j] < nums[minIdx]) {
                minIdx = j;
            }
        }
        swap(nums[i], nums[minIdx]);  // 交换到当前位置
    }
}
```

### 2. 冒泡排序
**特点**：稳定排序，原地操作，基础版时间复杂度 O(n²)，优化版最好 O(n)
```cpp
// 基础版
void bubbleSort(vector<int>& nums) {
    int n = nums.size();
    // 每轮将最大元素"冒泡"到末尾
    for (int i = n - 1; i > 0; i--) {
        for (int j = 0; j < i; j++) {
            if (nums[j] > nums[j + 1]) {
                swap(nums[j], nums[j + 1]);
            }
        }
    }
}

// 优化版（添加有序标记）
void bubbleSortOptimized(vector<int>& nums) {
    int n = nums.size();
    for (int i = n - 1; i > 0; i--) {
        bool isSorted = true;  // 标记本轮是否发生交换
        for (int j = 0; j < i; j++) {
            if (nums[j] > nums[j + 1]) {
                swap(nums[j], nums[j + 1]);
                isSorted = false;
            }
        }
        if (isSorted) break;  // 无交换说明已有序，直接退出
    }
}
```

### 3. 插入排序
**特点**：稳定排序，原地操作，适合小规模数据或接近有序数据，时间复杂度 O(n²)
```cpp
void insertionSort(vector<int>& nums) {
    int n = nums.size();
    // 从第二个元素开始插入
    for (int i = 1; i < n; i++) {
        int base = nums[i];  // 待插入元素
        int j = i - 1;       // 已排序区间末尾指针
        // 找到插入位置
        while (j >= 0 && nums[j] > base) {
            nums[j + 1] = nums[j];  // 元素后移
            j--;
        }
        nums[j + 1] = base;  // 插入待排序元素
    }
}
```

### 4. 快速排序
**特点**：不稳定排序，分治思想，平均时间复杂度 O(n log n)，最坏 O(n²)
```cpp
// 哨兵划分：以nums[left]为基准，划分左右子数组
int partition(vector<int> &nums, int left, int right) {
    int i = left, j = right;
    while (i < j) {
        // 从右向左找首个小于基准的元素
        while (i < j && nums[j] >= nums[left]) j--;
        // 从左向右找首个大于基准的元素
        while (i < j && nums[i] <= nums[left]) i++;
        swap(nums[i], nums[j]);  // 交换两个元素
    }
    swap(nums[i], nums[left]);  // 基准元素归位
    return i;  // 返回基准元素索引
}

// 快速排序主函数
void quickSort(vector<int> &nums, int left, int right) {
    if (left >= right) return;  // 递归终止条件
    int pivot = partition(nums, left, right);  // 划分
    quickSort(nums, left, pivot - 1);          // 左子数组排序
    quickSort(nums, pivot + 1, right);         // 右子数组排序
}

// 调用接口（简化使用）
void quickSort(vector<int> &nums) {
    quickSort(nums, 0, nums.size() - 1);
}
```

### 5. 归并排序
**特点**：稳定排序，分治思想，时间复杂度 O(n log n)，空间复杂度 O(n)
```cpp
// 合并两个有序子数组：[left, mid] 和 [mid+1, right]
void merge(vector<int> &nums, int left, int mid, int right) {
    vector<int> tmp(right - left + 1);  // 临时数组存储合并结果
    int i = left, j = mid + 1, k = 0;   // 三个指针

    // 合并两个子数组
    while (i <= mid && j <= right) {
        tmp[k++] = (nums[i] <= nums[j]) ? nums[i++] : nums[j++];
    }

    // 复制剩余元素
    while (i <= mid) tmp[k++] = nums[i++];
    while (j <= right) tmp[k++] = nums[j++];

    // 临时数组结果复制回原数组
    for (k = 0; k < tmp.size(); k++) {
        nums[left + k] = tmp[k];
    }
}

// 归并排序主函数
void mergeSort(vector<int> &nums, int left, int right) {
    if (left >= right) return;  // 递归终止条件
    int mid = left + (right - left) / 2;  // 划分中点
    mergeSort(nums, left, mid);           // 左子数组排序
    mergeSort(nums, mid + 1, right);      // 右子数组排序
    merge(nums, left, mid, right);        // 合并子数组
}

// 调用接口（简化使用）
void mergeSort(vector<int> &nums) {
    mergeSort(nums, 0, nums.size() - 1);
}
```

### 6. 堆排序
**特点**：不稳定排序，基于堆结构，时间复杂度 O(n log n)，空间复杂度 O(1)
```cpp
// 堆化操作：从节点i开始，自上而下维护大顶堆
void siftDown(vector<int> &nums, int n, int i) {
    while (true) {
        int left = 2 * i + 1;  // 左子节点索引
        int right = 2 * i + 2; // 右子节点索引
        int maxIdx = i;        // 最大值节点索引

        // 找到i、left、right中的最大值
        if (left < n && nums[left] > nums[maxIdx]) maxIdx = left;
        if (right < n && nums[right] > nums[maxIdx]) maxIdx = right;

        if (maxIdx == i) break;  // 已满足大顶堆，退出
        swap(nums[i], nums[maxIdx]);  // 交换节点
        i = maxIdx;  // 继续向下堆化
    }
}

// 堆排序主函数
void heapSort(vector<int> &nums) {
    int n = nums.size();
    // 建堆：从最后一个非叶节点开始堆化
    for (int i = n / 2 - 1; i >= 0; --i) {
        siftDown(nums, n, i);
    }

    // 提取最大值：循环n-1次
    for (int i = n - 1; i > 0; --i) {
        swap(nums[0], nums[i]);  // 根节点（最大值）移到末尾
        siftDown(nums, i, 0);    // 维护剩余元素为大顶堆
    }
}
```

## 分治算法
**核心思想**：将复杂问题分解为相似子问题，递归求解后合并结果，包含分解、解决、合并三步。

### 1. 分治算法模板
```cpp
// 分治函数模板（ReturnType需根据实际问题定义）
template <typename ReturnType>
ReturnType divideAndConquer(const vector<int>& input, int left, int right) {
    // 1. 递归终止条件（子问题足够小）
    if (left == right) {
        return input[left];  // 示例：返回单个元素（可根据需求修改）
    }

    // 2. 分解：分成左右两个子问题
    int mid = left + (right - left) / 2;

    // 3. 解决：递归求解子问题
    ReturnType leftResult = divideAndConquer<ReturnType>(input, left, mid);
    ReturnType rightResult = divideAndConquer<ReturnType>(input, mid + 1, right);

    // 4. 合并：合并子问题结果
    return mergeResults(leftResult, rightResult);
}

// 结果合并函数（示例：求最大值）
int mergeResults(int left, int right) {
    return max(left, right);
}
```

### 2. 分治-构建二叉树
**问题场景**：根据前序遍历和中序遍历序列构建二叉树
```cpp
// 二叉树节点定义
struct TreeNode {
    int val;
    TreeNode *left;
    TreeNode *right;
    TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
};

// 递归构建函数
TreeNode *buildTreeDFS(vector<int> &preorder, unordered_map<int, int> &inorderMap, 
                      int preIdx, int inLeft, int inRight) {
    // 终止条件：子树区间为空
    if (inLeft > inRight) return nullptr;

    // 根节点：前序遍历的当前元素
    int rootVal = preorder[preIdx];
    TreeNode *root = new TreeNode(rootVal);

    // 找到根节点在中序遍历中的位置，划分左右子树
    int rootInIdx = inorderMap[rootVal];
    int leftSubtreeSize = rootInIdx - inLeft;  // 左子树节点数

    // 递归构建左右子树
    root->left = buildTreeDFS(preorder, inorderMap, preIdx + 1, inLeft, rootInIdx - 1);
    root->right = buildTreeDFS(preorder, inorderMap, preIdx + 1 + leftSubtreeSize, rootInIdx + 1, inRight);

    return root;
}

// 主函数
TreeNode *buildTree(vector<int> &preorder, vector<int> &inorder) {
    // 构建中序遍历值到索引的映射，加速查找
    unordered_map<int, int> inorderMap;
    for (int i = 0; i < inorder.size(); i++) {
        inorderMap[inorder[i]] = i;
    }

    return buildTreeDFS(preorder, inorderMap, 0, 0, inorder.size() - 1);
}
```

## 回溯算法
**核心思想**：尝试所有可能解，递归搜索时记录状态，不满足条件则回溯（撤销选择），适用于组合、排列、子集等问题。

### 1. 回溯算法模板
```cpp
// 状态结构体（根据实际问题定义）
struct State {
    vector<int> path;  // 示例：当前选择路径
    int sum;           // 示例：当前总和
};

// 检查是否为有效解
bool isSolution(const State &state, int target) {
    return state.sum == target;  // 示例：总和等于目标值
}

// 记录解
void recordSolution(const State &state, vector<vector<int>> &res) {
    res.push_back(state.path);
}

// 检查选择是否合法（剪枝条件）
bool isValid(const State &state, int choice, int target) {
    return state.sum + choice <= target;  // 示例：不超过目标值
}

// 做出选择
void makeChoice(State &state, int choice) {
    state.path.push_back(choice);
    state.sum += choice;
}

// 撤销选择（回溯）
void undoChoice(State &state, int choice) {
    state.path.pop_back();
    state.sum -= choice;
}

// 回溯算法框架
void backtrack(State &state, const vector<int> &choices, int target, vector<vector<int>> &res) {
    // 检查是否为解
    if (isSolution(state, target)) {
        recordSolution(state, res);
        return;
    }

    // 遍历所有选择
    for (int choice : choices) {
        // 剪枝：跳过不合法选择
        if (isValid(state, choice, target)) {
            makeChoice(state, choice);    // 尝试选择
            backtrack(state, choices, target, res);  // 递归搜索
            undoChoice(state, choice);    // 回溯：撤销选择
        }
    }
}
```

### 2. 回溯-全排列问题
```cpp
vector<vector<int>> permute(vector<int> &nums) {
    vector<vector<int>> res;    // 存储所有排列结果
    vector<int> path;           // 当前排列路径
    vector<bool> used(nums.size(), false);  // 标记元素是否已使用

    // 回溯函数
    function<void()> backtrack = [&]() {
        // 终止条件：路径长度等于数组长度
        if (path.size() == nums.size()) {
            res.push_back(path);
            return;
        }

        // 遍历所有元素
        for (int i = 0; i < nums.size(); i++) {
            if (!used[i]) {  // 剪枝：跳过已使用的元素
                used[i] = true;
                path.push_back(nums[i]);  // 尝试选择
                backtrack();              // 递归搜索
                path.pop_back();          // 回溯
                used[i] = false;
            }
        }
    };

    backtrack();
    return res;
}
```

### 3. 回溯-子集和问题
**问题场景**：找出所有和为目标值的子集（允许重复选择元素）
```cpp
void subsetSumBacktrack(vector<int> &state, int target, int currentSum, 
                       int startIdx, const vector<int> &choices, vector<vector<int>> &res) {
    // 终止条件：当前和等于目标值
    if (currentSum == target) {
        res.push_back(state);
        return;
    }

    // 遍历选择（startIdx避免重复子集）
    for (int i = startIdx; i < choices.size(); i++) {
        int choice = choices[i];
        // 剪枝：超过目标值则跳过
        if (currentSum + choice > target) continue;

        state.push_back(choice);  // 尝试选择
        // 递归：允许重复选择当前元素，startIdx = i
        subsetSumBacktrack(state, target, currentSum + choice, i, choices, res);
        state.pop_back();         // 回溯
    }
}

// 主函数
vector<vector<int>> subsetSum(vector<int> &nums, int target) {
    vector<vector<int>> res;
    vector<int> state;
    // 排序便于剪枝（可选，优化性能）
    sort(nums.begin(), nums.end());
    subsetSumBacktrack(state, target, 0, 0, nums, res);
    return res;
}
```

## 动态规划(DP)
**核心思想**：将问题分解为子问题，存储子问题解避免重复计算，采用"自底向上"的递推方式，核心是状态定义、转移方程、边界条件。

### 1. 动态规划模板思路
1. **状态定义**：明确 dp[i] 或 dp[i][j] 表示的具体含义（关键步骤）
2. **转移方程**：找出 dp[i] 与 dp[i-1] 等前序状态的关系
3. **边界条件**：定义最小子问题的解（递推起点）
4. **递推顺序**：确保计算 dp[i] 时，所需的前序状态已计算完成

### 2. 动态规划-走楼梯
**问题描述**：每次可走1或2阶，求到第n阶的方法数
```cpp
// 基础版：无约束
int climbingStairs(int n) {
    if (n <= 2) return n;  // 边界条件
    vector<int> dp(n + 1, 0);
    dp[1] = 1;  // 1阶：1种方法
    dp[2] = 2;  // 2阶：2种方法
    // 转移方程：dp[i] = dp[i-1] + dp[i-2]
    for (int i = 3; i <= n; ++i) {
        dp[i] = dp[i-1] + dp[i-2];
    }
    return dp[n];
}

// 进阶版：不允许连续跳1阶
int climbingStairsWithConstraint(int n) {
    if (n <= 2) return n;  // 边界条件
    // dp[i][1]：第i阶以跳1阶结束；dp[i][2]：第i阶以跳2阶结束
    vector<vector<int>> dp(n + 1, vector<int>(3, 0));
    dp[1][1] = 1;
    dp[1][2] = 0;
    dp[2][1] = 0;
    dp[2][2] = 1;
    // 转移方程
    for (int i = 3; i <= n; ++i) {
        dp[i][1] = dp[i-1][2];  // 前一阶必须跳2阶
        dp[i][2] = dp[i-2][1] + dp[i-2][2];  // 前两阶可跳1或2阶
    }
    return dp[n][1] + dp[n][2];
}
```

### 3. 动态规划-01 背包问题
**问题描述**：每种物品最多选1个，求容量为cap的背包能装的最大价值
```cpp
int knapsack01(vector<int> &weights, vector<int> &values, int cap) {
    int n = weights.size();
    // dp[i][j]：前i个物品，容量j时的最大价值
    vector<vector<int>> dp(n + 1, vector<int>(cap + 1, 0));

    for (int i = 1; i <= n; ++i) {
        for (int j = 1; j <= cap; ++j) {
            if (j < weights[i-1]) {
                // 容量不足，不选第i个物品
                dp[i][j] = dp[i-1][j];
            } else {
                // 选或不选，取最大值
                dp[i][j] = max(dp[i-1][j], 
                              dp[i-1][j - weights[i-1]] + values[i-1]);
            }
        }
    }
    return dp[n][cap];
}

// 空间优化版（O(cap)空间）
int knapsack01Optimized(vector<int> &weights, vector<int> &values, int cap) {
    int n = weights.size();
    vector<int> dp(cap + 1, 0);

    for (int i = 0; i < n; ++i) {
        // 逆序遍历，避免覆盖前序状态
        for (int j = cap; j >= weights[i]; --j) {
            dp[j] = max(dp[j], dp[j - weights[i]] + values[i]);
        }
    }
    return dp[cap];
}
```

### 4. 动态规划-零钱兑换
**问题描述**：用给定硬币凑成目标金额，求最少硬币数（硬币可重复使用）
```cpp
int coinChange(vector<int> &coins, int amount) {
    int MAX = amount + 1;
    // dp[i]：凑成金额i所需的最少硬币数
    vector<int> dp(amount + 1, MAX);
    dp[0] = 0;  // 边界条件：金额0需0个硬币

    for (int i = 1; i <= amount; ++i) {
        for (int coin : coins) {
            if (coin <= i) {
                // 转移方程：取"不选当前硬币"和"选当前硬币"的最小值
                dp[i] = min(dp[i], dp[i - coin] + 1);
            }
        }
    }
    // 若dp[amount]仍为MAX，说明无法凑成
    return dp[amount] == MAX ? -1 : dp[amount];
}
```

## 贪心算法
**核心思想**：每个决策阶段选择当前最优解（局部最优），最终期望得到全局最优解。关键是证明"贪心选择性质"和"最优子结构"。

### 1. 贪心算法模板
```cpp
// 通用贪心模板（排序+贪心选择）
int greedyTemplate(vector<pair<int, int>>& items) {
    // 1. 按贪心策略排序（核心步骤，策略需根据问题定义）
    sort(items.begin(), items.end(), [](const pair<int, int>& a, const pair<int, int>& b) {
        // 示例：按第二个元素升序（如活动选择问题的结束时间）
        return a.second < b.second;
    });

    // 2. 遍历并执行贪心选择
    int result = 0;
    int last = -1;  // 记录上一次选择的状态（如最后一个活动的结束时间）
    for (auto& item : items) {
        // 贪心选择条件（根据问题定义）
        if (item.first > last) {
            result++;
            last = item.second;
        }
    }

    return result;
}
```

### 2. 贪心-分数背包
**问题描述**：物品可分割，求容量为cap的背包能装的最大价值
```cpp
// 物品结构体
struct Item {
    int weight;
    int value;
    Item(int w, int v) : weight(w), value(v) {}
};

double fractionalKnapsack(vector<int> &weights, vector<int> &values, int cap) {
    vector<Item> items;
    for (int i = 0; i < weights.size(); ++i) {
        items.emplace_back(weights[i], values[i]);
    }

    // 贪心策略：按单位价值（value/weight）降序排序
    sort(items.begin(), items.end(), [](const Item& a, const Item& b) {
        return (double)a.value / a.weight > (double)b.value / b.weight;
    });

    double maxValue = 0.0;
    for (auto& item : items) {
        if (item.weight <= cap) {
            // 容量充足，全取
            maxValue += item.value;
            cap -= item.weight;
        } else {
            // 容量不足，取部分
            maxValue += (double)item.value / item.weight * cap;
            break;  // 背包已满
        }
    }

    return maxValue;
}
```

### 3. 贪心-Dijkstra 算法
**问题描述**：求带权无向图中从源点到所有其他节点的最短路径（边权非负）
```cpp
#include <iostream>
#include <vector>
#include <climits>
#include <algorithm>
using namespace std;

const int INF = INT_MAX / 2;  // 避免溢出
const int MAX_V = 100;        // 最大顶点数

int main() {
    int V, E;  // 顶点数、边数
    cin >> V >> E;

    // 邻接矩阵存储图（初始化为INF）
    vector<vector<int>> graph(V, vector<int>(V, INF));
    for (int i = 0; i < V; ++i) {
        graph[i][i] = 0;  // 自身到自身距离为0
    }

    // 读取边权
    for (int i = 0; i < E; ++i) {
        int u, v, w;
        cin >> u >> v >> w;
        graph[u][v] = w;
        graph[v][u] = w;  // 无向图
    }

    int start;  // 源点
    cin >> start;

    vector<int> dist(V, INF);  // 源点到各点的最短距离
    vector<bool> visited(V, false);  // 标记是否已确定最短路径
    dist[start] = 0;

    while (true) {
        // 1. 选择未访问节点中距离最小的节点
        int u = -1;
        for (int i = 0; i < V; ++i) {
            if (!visited[i] && (u == -1 || dist[i] < dist[u])) {
                u = i;
            }
        }

        if (u == -1) break;  // 所有节点已访问
        visited[u] = true;   // 标记为已确定

        // 2. 松弛操作：更新通过u到达其他节点的距离
        for (int v = 0; v < V; ++v) {
            if (!visited[v] && graph[u][v] != INF) {
                dist[v] = min(dist[v], dist[u] + graph[u][v]);
            }
        }
    }

    // 输出结果
    for (int i = 0; i < V; ++i) {
        cout << "源点 " << start << " 到节点 " << i << " 的最短距离：";
        if (dist[i] == INF) cout << "不可达";
        else cout << dist[i];
        cout << endl;
    }

    return 0;
}
```
